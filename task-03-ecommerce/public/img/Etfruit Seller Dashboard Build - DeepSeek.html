You are 100% right, bro. Looking at raw UUIDs (c4ccf4f7...) is impossible to read. This is a classic database visualization problem.

Here is the reality: Prisma Studio is just showing you the raw database. It stores IDs (UUIDs) because that's how relational databases link tables together. It doesn't automatically translate them to names to save space and speed up the database.

Here are the two ways to solve this: How to read it right now in Prisma Studio, and How to build it professionally in your Admin Dashboard.

1. How to see it in Prisma Studio right now

You don't need to change any code. Prisma Studio has a built-in feature for this.

  1. Look at the productId or sellerId columns in your screenshot.

  2. See that little → (arrow) next to the UUID?

  3. Click that arrow. It will instantly open a side-panel showing you the exact Product name (e.g., "Cauliflower Head") or Seller name (e.g., "Abebe Market").

  4. Alternatively, click on the Product or Seller table in the left sidebar. You will see a column called offers, and you can click the arrow there to see all the offers linked to that product.

2. The Professional Fix (How to show Names in the App)

You shouldn't have to use Prisma Studio to know who is selling what. Your Admin Dashboard or Seller Dashboard should show the names.

To do this, you need to tell Prisma to "Join" the tables using the include keyword.

Here is exactly how to write the backend function for the Admin or Seller to see a clear list of Offers with names:

The Backend (src/controllers/offerController.js or adminController.js)

// Fetch all offers and include the Product Name and Seller Name
exports.getAllOffers = asyncHandler(async (req, res) => {
  const offers = await prisma.offer.findMany({
    include: {
      // Bring in the Product table
      product: {
        select: { name: true, category: true, image: true }
      },
      // Bring in the Seller table
      seller: {
        select: { businessName: true, city: true, verificationStatus: true }
      }
    },
    orderBy: { createdAt: 'desc' }
  });

  res.json({ success: true, data: offers });
});

import { API } from './api.js';

const formatETB = (cents) => (cents / 100).toLocaleString('en-ET', { minimumFractionDigits: 2 }) + ' ETB';

async function loadCart() {
  const container = document.getElementById('cart-items-container');
  const summary = document.getElementById('cart-summary');
  const countEl = document.getElementById('cart-item-count');

  try {
    const res = await API.get('/api/cart');
    const cart = res.data;

    // 1. Handle Empty Cart
    if (!cart || !cart.items || cart.items.length === 0) {
      container.innerHTML = `
        <div class="orders-empty">
          <div class="orders-empty-icon">🛒</div>
          <h2>Your cart is empty</h2>
          <p>Looks like you haven't added any fresh produce yet.</p>
          <a href="/" class="btn btn-amber">Start Shopping</a>
        </div>
      `;
      summary.style.display = 'none';
      if (countEl) countEl.textContent = '0 items';
      return;
    }

    // 2. Render Items
    let subtotal = 0;
    if (countEl) countEl.textContent = `${cart.items.length} items`;

    container.innerHTML = cart.items.map(item => {
      const offer = item.offer;
      const product = offer.product;
      const lineTotal = offer.pricePerUnit * item.quantity;
      subtotal += lineTotal;

      return `
        <div class="cart-row">
          <div class="cart-row-img">
            <img src="${product.image}" alt="${product.name}">
          </div>
          <div class="cart-row-info">
            <span class="cart-row-cat">${product.category}</span>
            <span class="cart-row-name">${product.name}</span>
            <span class="cart-row-merchant">Sold by: 🚜 ${offer.seller.businessName}</span>
          </div>
          <div class="cart-row-qty">
            <span style="font-weight: 600; color: var(--text);">${item.quantity} ${offer.unit}</span>
          </div>
          <div class="cart-row-total">
            ${formatETB(lineTotal)}
          </div>
          <button class="cart-row-remove" onclick="removeItem('${item.id}')">×</button>
        </div>
      `;
    }).join('');

    // 3. Update Summary
    summary.style.display = 'block';
    document.getElementById('cart-subtotal').textContent = formatETB(subtotal);
    document.getElementById('cart-total').textContent = formatETB(subtotal + 5000); // + 50 ETB Delivery

  } catch (err) {
    container.innerHTML = `<p style="color:red; text-align:center;">Failed to load cart. Please log in.</p>`;
  }
}

window.removeItem = async (itemId) => {
  try {
    await API.delete(`/api/cart/remove/${itemId}`); // Assuming you have this backend route
    loadCart(); // Reload
  } catch (err) {
    alert('Failed to remove item');
  }
};

document.addEventListener('DOMContentLoaded', loadCart);

import { API } from './api.js';

const formatETB = (cents) => (cents / 100).toLocaleString('en-ET', { minimumFractionDigits: 2 }) + ' ETB';

async function loadOrders() {
  const container = document.getElementById('orders-list');

  try {
    const res = await API.get('/api/orders');
    const orders = res.data;

    if (!orders || orders.length === 0) {
      container.innerHTML = `
        <div class="orders-empty">
          <div class="orders-empty-icon">📦</div>
          <h2>No orders yet</h2>
          <p>When you place an order, it will appear here.</p>
          <a href="/" class="btn btn-amber">Start Shopping</a>
        </div>
      `;
      return;
    }

    container.innerHTML = orders.map(order => {
      // Format Date
      const date = new Date(order.createdAt).toLocaleDateString('en-US', { 
        month: 'short', day: 'numeric', year: 'numeric' 
      });

      // Status styling class
      const statusClass = `order-status-${order.status}`;

      return `
        <div class="order-card" onclick="window.location.href='/order-detail.html?id=${order.id}'" style="cursor: pointer;">
          <div class="order-card-head">
            <div>
              <div class="order-card-id">${order.id.slice(0, 8).toUpperCase()}</div>
              <div class="order-card-date">${date}</div>
            </div>
            <span class="order-status ${statusClass}">${order.status.replace('_', ' ').toUpperCase()}</span>
          </div>

          <div class="order-card-body">
            <div class="order-thumbs">
              <!-- Show up to 2 images -->
              <div class="order-thumb"><img src="${order.items[0]?.offer?.product?.image || '/img/placeholder.png'}"></div>
            </div>
            <div class="order-items-text">
              <div class="order-items-title">
                ${order.items[0]?.offer?.product?.name || 'Item'} 
                ${order.items.length > 1 ? `+ ${order.items.length - 1} more` : ''}
              </div>
              <div class="order-items-sub">Delivered to: ${order.shippingAddress || 'Addis Ababa'}</div>
            </div>
            <div class="order-card-total">
              <div class="order-total-label">Total</div>
              <div class="order-total-value">${formatETB(order.totalAmount)}</div>
            </div>
          </div>
          
          <div class="order-card-foot">
            <span class="order-address">🛡️ Paid via ${order.paymentMethod || 'Chapa'}</span>
            <button class="order-btn order-btn-outline">View Details →</button>
          </div>
        </div>
      `;
    }).join('');

  } catch (err) {
    container.innerHTML = `<p style="color:red; text-align:center;">Failed to load orders. Please log in.</p>`;
  }
